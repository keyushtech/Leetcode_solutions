// Title: Single Element in a Sorted Array
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/single-element-in-a-sorted-array/

class Solution:
    def singleNonDuplicate(self, nums: List[int]) -> int:
        def binary_search(nums,start,end):
            if start<=end:
                mid = (end-start) //2 + start
                if len(nums) ==1:
                if mid==0 and nums[0]!= nums[1] :
                    return nums[mid]
                if mid == len(nums)-1 and nums[len(nums)-1] != nums[len(nums)-2]:
                    return nums[mid]
                if nums[mid+1] != nums[mid] and nums[mid] != nums[mid - 1]:
                    return nums[mid]
                if (mid%2)==0:
                    if nums[mid] == nums[mid-1]:
                        return binary_search(nums,start,mid-1)
                    else:
                        return binary_search(nums,mid+1,end)

                else:
                    if nums[mid] == nums[mid-1]:
                        return binary_search(nums,mid+1,end)
                    else:
                        return binary_search(nums,start,mid)
            return -1
                    return nums[0]
        return binary_search(nums,0,len(nums)-1)
