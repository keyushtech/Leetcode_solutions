// Title: Reverse Prefix of Word
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/reverse-prefix-of-word/

class Solution:
    def reversePrefix(self, word: str, ch: 
    str) -> str:
        idx = word.find(ch)
        if idx == -1:
            return word
        stack = []
        for i in range(idx + 1):
            stack.append(word[i])
        res = []
        while stack:
            res.append(stack.pop())
        res.append(word[idx + 1:])
        return "".join(res)
