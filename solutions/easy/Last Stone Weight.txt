// Title: Last Stone Weight
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/last-stone-weight/

class Solution:
    def lastStoneWeight(self, stones: list[int]) -> int:
        while len(stones)>1:
            largest = max(stones)
            stones.remove(largest)
            largest2 = max(stones)
            stones.remove(largest2)
            diff = largest - largest2
            stones.append(diff)
        if len(stones)==1:
            return stones[0]
        else:
            return 0
