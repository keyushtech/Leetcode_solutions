// Title: Last Stone Weight
            // Difficulty: Easy
            // Language: Python3

class Solution:
    def lastStoneWeight(self, stones: list[int]) -> int:
        while len(stones)>1:
            stone1 = max(stones)
            stones.remove(max(stones))
            stone2 = max(stones)
            stones.remove(max(stones))
            if stone1 > stone2:
                diff = stone1 - stone2
                stones.append(diff)
        if len(stones)==1:

            return stones[0]
        else:
            return 0
