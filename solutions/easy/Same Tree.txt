class Solution:
    def isSameTree(self, p: Optional[TreeNode], q: 
        if not p and not q:
            return True
        if not p or not q or p.val != q.val:
            return False
        return self.isSameTree(p.left, q.left) and self.
    Optional[TreeNode]) -> bool:
        isSameTree(p.right, q.right)
