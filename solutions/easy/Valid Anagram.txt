// Title: Valid Anagram
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/valid-anagram/

class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        s = list(s)
        t = list(t)
        for i in s:
            if i in t:
                t.remove(i)
            else:
                return False
        if len(s)!=len(t):
            return False
        return True
